## 二、完整可运行代码 · 论文完整草稿完成度追踪器（+ 预期输出）

> 这个脚本追踪论文完整草稿的完成度——各章节 + 图表 + 实验 + 自审 + review 发送，输出完成度和"还差什么"，帮你看清完整草稿是否齐全、能否发 review。

```python
#!/usr/bin/env python3
# paper_draft_completion_tracker.py —— 论文完整草稿完成度追踪器
chapters = {  # 所有章节
    "Abstract": True, "Introduction": True, "Related Work": True,
    "Method": True, "Experiment": True, "Conclusion": True, "参考文献": True,
}
assets = {  # 图表+实验+串联
    "所有图表(清晰/统一/突出卖点)": True, "所有实验(完整结果)": True,
    "全文串联(逻辑/术语/过渡)": True,
}
self_review = {  # 自审
    "通读(读者视角)": True, "mock review(审稿人视角)": True, "据自审完善": True,
}
review_sent = False  # 发给所有合作者review

def rate(d):
    done = sum(1 for v in d.values() if v); return done, len(d), round(done/len(d)*100)

def track():
    print("=== 论文完整草稿完成度(W67)===\n")
    all_done = True
    for name, d in [("所有章节", chapters), ("图表+实验+串联", assets), ("自审", self_review)]:
        done, tot, pct = rate(d)
        print(f"{name}: {done}/{tot} = {pct}%")
        for k, v in d.items():
            if not v:
                print(f"   □ 待办: {k}"); all_done = False
    print(f"\n发合作者review: {'✅ 已发(给1周)' if review_sent else '⏳ 待发(周日发给所有合作者)'}")
    print("\n" + "="*42)
    if all_done:
        print("🎉 完整草稿齐全(所有章节+图表+实验+自审完善)!")
        if not review_sent:
            print("   → 周日发给所有合作者(老师+师兄)review,给1周时间。")
        else:
            print("   → 已发review,W68收反馈逐条修改→定稿→选投稿目标。")
    else:
        print("→ 补齐未完成项,再发合作者review。")

if __name__ == "__main__":
    track()
```

```

所有章节: 7/7 = 100%
图表+实验+串联: 3/3 = 100%
自审: 3/3 = 100%

发合作者review: ⏳ 待发(周日发给所有合作者)

==========================================
🎉 完整草稿齐全(所有章节+图表+实验+自审完善)!
   → 周日发给所有合作者(老师+师兄)review,给1周时间。
```
> 解读：把完整草稿的各部分状态填进去，脚本算出完成度。示例里所有章节、图表+实验+串联、自审都完成（完整草稿齐全），就差周日发给所有合作者 review。它把"完整草稿是否齐全、能否发 review"一目了然——确保你发给合作者的是一份完整、自审过的草稿。
