## 二、完整可运行代码 · 多场面试+三线追踪器（+ 预期输出）

> 这个脚本帮你追踪面试密集期：多场面试状态+三线进度+M20 硬交付。复制到 `multi_interview_tracker.py` 运行。

```python
# multi_interview_tracker.py —— W81面试密集期+三线追踪器
# 用法: python multi_interview_tracker.py

# ============ 多场面试(本周) ============
interviews = [
    # (单位, 状态, 复盘, 感谢邮件)
    ("★空天院",  "已面试", True, True),
    ("自动化所",  "已面试", True, True),
    ("武大遥感",  "约面试(周六)", False, False),
]

# ============ 三线进度 ============
three_lines = {
    "面试线(本周重心)": "2场已面试+1场约定、每场复盘",
    "博士线(见缝插针)": "文献综述章完成✓ + 方法章第3章起步",
    "遥感线(按需)": "检查论文状态、第二篇返修中午段处理",
}

# ============ M20硬交付(累计) ============
m20_deliverables = {
    "面试题库完善(40+题+标准回答)": True,
    "模拟面试≥3次": True,   # W78第1次+W79第2次+...
    "正式面试≥2次": True,   # 本周多场
    "面试纪要+感谢邮件": True,
    "博士论文引言+文献综述完成": True,  # 本周文献综述完成
}

def bar(d, t):
    n = int(20*min(d,t)/t) if t else 0
    return "█"*n + "░"*(20-n) + f" {d}/{t}"

if __name__ == "__main__":
    print("="*56, "\n🎯 W81面试密集期+三线追踪\n" + "="*56)

    print("\n【多场面试(本周)】")
    interviewed = sum(1 for _,s,_,_ in interviews if "已面试" in s)
    for unit, status, review, thanks in interviews:
        r = "✓复盘" if review else "待复盘"
        t = "✓感谢" if thanks else "待感谢"
        print(f"  · {unit}: {status} | {r} | {t}")
    print(f"  已面试: {interviewed}场 (正式面试M20目标≥2次)")

    print("\n【三线并行】")
    for line, status in three_lines.items():
        print(f"  · {line}: {status}")

    print("\n【M20硬交付(累计)】")
    done = sum(m20_deliverables.values())
    for k, v in m20_deliverables.items():
        print(f"  [{'✓' if v else ' '}] {k}")
    print(f"  达成度: {bar(done, len(m20_deliverables))}",
          "✓ 接近M20收官!" if done>=len(m20_deliverables)-1 else "")

    print("\n" + "="*56)
    print("  ★面试为主+博士见缝插针+遥感按需、三线并行不乱")
    print("  ★多场面试每场全力+复盘成经验、为M21空天院面试准备")
    print("="*56)
```
