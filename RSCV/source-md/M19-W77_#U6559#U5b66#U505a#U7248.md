## 二、完整可运行代码 · 面试题库管理器（+ 预期输出）

> 这个脚本帮你管理面试准备：4 大类题目准备进度 + 薄弱点 + 模拟面试就绪度。复制到 `interview_prep.py` 运行。

```python
# interview_prep.py —— 面试题库准备管理器
# 用法: python interview_prep.py

# ============ 4大类准备进度 ============
# (类别, 目标题数, 已准备, 薄弱点)
categories = [
    ("科研问题(讲透3篇论文+深挖)", 10, 8, "第3篇论文的深挖追问"),
    ("技术基础(深度学习/CV/遥感)", 10, 7, "遥感SAR基础概念"),
    ("编程能力(算法题+手写)",      10, 5, "动态规划、手写attention"),
    ("个人问题(★转型者关键)",      10, 9, ""),
]

# ============ 模拟面试就绪度 ============
mock_ready = {
    "科研问题能讲透(有层次+应对追问)": True,
    "技术基础扎实(不只自己方向)": False,  # 薄弱
    "编程能保持手感": False,              # 薄弱
    "★为什么转方向(讲成优势)": True,
    "★为什么选空天院(具体)": True,
    "职业规划清晰": True,
}

if __name__ == "__main__":
    print("="*58, "\n🎤 面试题库准备管理\n" + "="*58)

    print("\n【4大类准备进度】(目标每类10题)")
    total_target = sum(c[1] for c in categories)
    total_done = sum(c[2] for c in categories)
    for cat, target, done, weak in categories:
        n = int(12*done/target)
        bar = "█"*n + "░"*(12-n)
        w = f" ⚠薄弱:{weak}" if weak else " ✓"
        print(f"  {cat}")
        print(f"    {bar} {done}/{target}{w}")
    print(f"\n  总进度: {total_done}/{total_target}题")

    print("\n【模拟面试就绪度】")
    ready = sum(mock_ready.values())
    for k, v in mock_ready.items():
        print(f"  [{'✓' if v else '⚠'}] {k}")
    pct = int(100*ready/len(mock_ready))
    print(f"  就绪度: {ready}/{len(mock_ready)} ({pct}%)")
    if ready < len(mock_ready):
        weak_items = [k for k,v in mock_ready.items() if not v]
        print(f"  → 加强: {', '.join(weak_items)}")

    print("\n" + "="*58)
    print("  下周(W78): 每类完善到10题 + 第一次模拟面试(找朋友)")
    print("  ★模拟流程: 自我介绍5min→科研10min→技术10min→个人5min→反馈15min")
    print("="*58)
```
