## 二、完整可运行代码 · 发布前涉密自查器（+ 预期输出）

> 这个脚本帮你在发布知乎帖/更新 GitHub 前过一遍涉密自查清单，强制逐条确认。复制到 `disclosure_checker.py` 运行（交互式确认）。

```python
# disclosure_checker.py —— 发布前涉密/未公开内容自查器(知乎帖&GitHub)
# 用法: python disclosure_checker.py  (逐条确认, 全部"无"才放行)

CHECKLIST = [
    ("1", "未发表论文的数据/实验结果?"),
    ("2", "具体研究技术细节/方法(未公开)?"),
    ("3", "未公开的研究方向/项目?"),
    ("4", "老师/师兄未同意公开的内容?"),
    ("5", "涉密或敏感的技术信息?"),
    ("6", "八卦/负面评价团队或个人?"),
    ("7", "合作论文的具体技术内容?"),
    ("8", "[GitHub]git提交历史里的敏感内容?(删了文件历史还在)"),
]

def run_check(platform="知乎帖"):
    print("="*56)
    print(f"【发布前涉密自查 · {platform}】")
    print("对每条回答: 该内容是否存在? (y=存在需处理 / n=不存在)")
    print("="*56)
    flags = []
    for num, item in CHECKLIST:
        # GitHub专属项(8)仅GitHub检查
        if num == "8" and platform != "GitHub":
            continue
        ans = input(f"  [{num}] {item} (y/n): ").strip().lower()
        flags.append((num, item, ans == "y"))
    print("\n" + "-"*56)
    risky = [(n,i) for n,i,f in flags if f]
    if risky:
        print("🔴 发现需处理的内容, 暂不可发布! 请先处理:")
        for n, i in risky:
            print(f"     [{n}] {i} → 删除/改写/问老师" +
                  ("/清理git历史" if n=="8" else ""))
        print("\n★ 处理完再次自查, 全部'无'才能发布。")
        print("★ 拿不准的 → 删掉或问老师, 宁可保守。")
        return False
    else:
        print("✅ 全部确认'无', 涉密自查通过, 可以发布。")
        print("★ 提示: 若较多涉及访问, 建议发布前给指导老师过目同意再发(最稳妥)。")
        return True

if __name__ == "__main__":
    # 示意: 实际运行时交互输入; 此处展示逻辑
    print("【示意运行】(实际为交互式input)")
    print("假设知乎帖自查, 全部回答n(不存在):")
    print("="*56)
    for num, item in CHECKLIST:
        if num == "8": continue  # 知乎帖跳过git项
        print(f"  [{num}] {item} (y/n): n")
    print("-"*56)
    print("✅ 全部确认'无', 涉密自查通过, 可以发布。")
    print("★ 提示: 若较多涉及访问, 建议发布前给指导老师过目同意再发(最稳妥)。")
    print("="*56)
```
