## 二、完整可运行代码 · offer 决策矩阵计算器（+ 预期输出）

> 这个脚本帮你用决策矩阵量化评估多个 offer。复制到 `offer_decision_matrix.py`，把分数改成你的实际打分后运行。

```python
# offer_decision_matrix.py —— W88 offer决策矩阵计算器
# 用法: 修改下面的scores为你的实际打分, 然后 python offer_decision_matrix.py

# ============ 维度和权重 ============
dimensions = [
    ("研究方向匹配度", 4),
    ("导师/团队实力",  4),
    ("职业发展前景",   3),
    ("薪资待遇",       2),
    ("城市/生活质量",  2),
    ("学术自由度",     2),
    ("合同灵活性",     1),
]
# 满分 = sum(权重)*10
MAX = sum(w for _, w in dimensions) * 10  # = 180

# ============ 各offer打分(1-10, 示例数据, 改成你的实际打分) ============
scores = {
    "空天院":    [9, 9, 8, 7, 7, 7, 7],   # 研究契合/团队强/发展好
    "Plan B-1":  [7, 7, 7, 8, 8, 7, 8],   # 待遇/城市好
    "Plan B-2":  [6, 6, 7, 9, 9, 6, 9],   # 待遇/城市更好但研究契合低
}

def weighted_total(score_list):
    return sum(s * w for s, (_, w) in zip(score_list, dimensions))

if __name__ == "__main__":
    print("="*60, "\n🎯 W88 offer决策矩阵\n" + "="*60)

    # 表头
    print(f"\n{'维度':<16}{'权重':<6}", end="")
    for name in scores:
        print(f"{name:<10}", end="")
    print()

    # 各维度打分
    for i, (dim, w) in enumerate(dimensions):
        print(f"{dim:<16}×{w:<5}", end="")
        for name in scores:
            print(f"{scores[name][i]:<10}", end="")
        print()

    # 加权总分
    print("-"*60)
    print(f"{'加权总分(/' + str(MAX) + ')':<22}", end="")
    totals = {}
    for name in scores:
        t = weighted_total(scores[name])
        totals[name] = t
        print(f"{t:<10}", end="")
    print()

    # 推荐
    best = max(totals, key=totals.get)
    print("\n" + "="*60)
    print(f"  📊 决策矩阵评分最高: {best} ({totals[best]}/{MAX})")
    ranked = sorted(totals.items(), key=lambda x: -x[1])
    print("  排序:", " > ".join(f"{n}({t})" for n, t in ranked))
    print("  ★评分是重要参考、结合你的实际偏好和直觉做最终决策")
    print("  ★空天院圆满最好、去其他单位/企业也是转型成功")
    print("="*60)
```
