#!/usr/bin/env python3
# m16_deliverable_tracker.py —— M16 硬交付进度追踪器
M16_deliverables = {

    done = sum(1 for v in d.values() if v["done"])
    total = len(d)
    for item, v in d.items():
        mark = "✅" if v["done"] else "⏳"
        print(f"{mark} {item}")
        print(f"     → {v['note']}")

        print("\n→ W65 收官前补齐(尤其组会30min汇报)。")

        print("\n🎉 M16 硬交付全部达成！可写 M16 月度复盘、准备 M17。")
