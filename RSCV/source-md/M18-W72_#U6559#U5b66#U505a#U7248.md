## 二、完整可运行代码 · 求职材料就绪检查器（+ 预期输出）

> 这个脚本帮你盘点阶段四求职"弹药"是否备齐：三件套 + 推荐信 + 论文成果 + 其他。复制到 `job_materials_checker.py` 运行。

```python
# job_materials_checker.py —— 阶段四求职材料就绪检查器
# 用法: python job_materials_checker.py

# ============ 1. 求职材料三件套 ============
three_pieces = {
    "简历(中文最新版)":   True,
    "简历(英文/CV最新版)": True,
    "个人主页(更新上线)":  True,
    "作品集PDF(≤5MB最新)": True,
}

# ============ 2. 推荐信(求职通常需2-3封) ============
NEED_LETTERS = 2
recommenders = [
    # (推荐人, 状态: 已提交/在写/待跟进/婉拒)
    ("空天院[老师A]", "在写"),
    ("[备选:资深师兄/另一老师]", "待跟进"),
]

# ============ 3. 论文成果 ============
papers = [
    ("第一篇(独作)", "已发表/在审"),
    ("第二篇(独作)", "在审"),
    ("第三篇(空天院合作)", "在审/已投"),
]

# ============ 4. 其他弹药 ============
others = {
    "竞赛排名(Top X%)": True,
    "开源GitHub(N stars)": True,
    "成果摘要(一页纸)": True,
}

def section(title, items_done, items_total):
    n = int(20*items_done/items_total) if items_total else 0
    return f"{title}: {'█'*n}{'░'*(20-n)} {items_done}/{items_total}"

if __name__ == "__main__":
    print("="*56, "\n【阶段四求职·弹药就绪检查】")

    print("\n[1. 求职材料三件套]")
    d1 = sum(three_pieces.values())
    for k,v in three_pieces.items():
        print(f"  [{'✓' if v else ' '}] {k}")
    print("  " + section("三件套", d1, len(three_pieces)))

    print("\n[2. 推荐信](求职需≥%d封)" % NEED_LETTERS)
    submitted = sum(1 for _,s in recommenders if s in ("已提交","在写"))
    for r,s in recommenders:
        flag = "✓" if s in ("已提交","在写") else ("⏳" if s=="待跟进" else "✗")
        print(f"  [{flag}] {r}: {s}")
    status = "✅落实在望" if submitted>=NEED_LETTERS else f"⚠️还需跟进({submitted}/{NEED_LETTERS})"
    print(f"  推荐信: {status}")

    print("\n[3. 论文成果]")
    for p,s in papers:
        print(f"  · {p}: {s}")
    print(f"  共 {len(papers)} 篇 (独立+合作, 叙事完整)")

    print("\n[4. 其他弹药]")
    d4 = sum(others.values())
    for k,v in others.items():
        print(f"  [{'✓' if v else ' '}] {k}")

    # 总体就绪判断
    print("\n【总体就绪】")
    ready = (d1==len(three_pieces)) and (submitted>=NEED_LETTERS) and (d4==len(others))
    if ready:
        print("  🎉 阶段四求职弹药备齐, 可启动投递!")
    else:
        gaps = []
        if d1<len(three_pieces): gaps.append("三件套未全就绪")
        if submitted<NEED_LETTERS: gaps.append(f"推荐信还差{NEED_LETTERS-submitted}封")
        if d4<len(others): gaps.append("其他弹药未全")
        print("  ⚠️ 还需补齐:", "; ".join(gaps), "→ 排进W73-74/阶段四")
    print("="*56)
```
