    "结果预案": "offer→M22决策 / 被拒→分析+Plan B",

    "面试纪要+感谢邮件": True,
    "Plan B面试≥1次(如有机会)": None,  # 视机会
    "博士论文~40%(引言+综述+方法+实验)": False,  # 30%→向40%
}

def bar(pct):
    n = int(20*pct/100)
    return "█"*n + "░"*(20-n) + f" {pct}%"

if __name__ == "__main__":
    print("="*58, "\n📖 W85博士论文进度+等结果追踪\n" + "="*58)

    print("\n【博士论文章节进度】")
    total = sum(c[2] for c in chapters)
    overall = total / (len(chapters)*100) * 100
    for name, status, pct in chapters:
        print(f"  {name}: {bar(pct)} ({status})")
    # 估算: 前4章为主体, 第5章占比小
    main_progress = sum(c[2] for c in chapters[:4]) / 400 * 100
    print(f"  → 主体(前4章)约: {main_progress:.0f}%; 整体博士论文约: 30%")

    print("\n【等结果】")
    for k, v in result_wait.items():
        print(f"  · {k}: {v}")

    print("\n【M21硬交付】")
    for k, v in m21_deliverables.items():
        mark = "✓" if v is True else ("…" if v is None else " ")
        print(f"  [{mark}] {k}")

    print("\n" + "="*58)
    print("  ★等结果期: 博士论文加速到30%(核心框架就绪)+多手准备")
    print("  ★W86: 实验章完成+遥感维护; W87: 总结展望章+M21月复盘")
    print("="*58)
