## 二、完整可运行代码 · 访问后半冲刺追踪器（+ 预期输出）

> 这个脚本追踪访问后半冲刺的核心进度——剩余实验完成度 + 论文章节完成度 + 署名状态 + 组会汇报累计 + 博士维护，输出冲刺进度和"还差什么"，帮你看清离完整草稿和访问收官还差多少。

```python
#!/usr/bin/env python3
# visit_sprint_tracker.py —— 访问后半冲刺追踪器
experiments = {  # 核心实验最后一轮(访问结束前跑完所有实验)
    "baseline对比(完整)": True, "消融(完整)": True, "可视化(完整)": True,
    "据1v1补的实验": True, "所有实验做扎实+归档": False,
}
paper_chapters = {  # 合作论文各章节(W67目标:完整草稿)
    "Method": True, "Experiment": True, "Introduction": False,
    "Related Work": False, "Conclusion": False, "Abstract": False, "参考文献": False,
}
signoff = {"署名定(一作/通讯/投哪)": True, "投稿目标明确": True}
group_reports_total = 2  # 组会汇报累计(M17硬交付要≥2次)

def rate(d):
    done = sum(1 for v in d.values() if v); return done, len(d), round(done/len(d)*100)

def track():
    print("=== 访问后半冲刺进度(W66)===\n")
    for name, d in [("核心实验最后一轮", experiments), ("合作论文章节(W67完整草稿)", paper_chapters),
                    ("署名", signoff)]:
        done, tot, pct = rate(d)
        print(f"{name}: {done}/{tot} = {pct}%")
        for k, v in d.items():
            if not v:
                print(f"   □ 待办: {k}")
    print(f"\n组会汇报累计: {group_reports_total} 次 (M17硬交付要≥2次){' ✅' if group_reports_total>=2 else ' ⏳'}")
    print("\n" + "="*42)
    e_pct = rate(experiments)[2]; p_pct = rate(paper_chapters)[2]
    if e_pct >= 80 and signoff["署名定(一作/通讯/投哪)"]:
        print(f"实验{e_pct}% / 论文章节{p_pct}% / 署名已定")
        print("→ W67冲刺完整草稿(剩余章节:Intro/RW/Conclusion/Abstract/参考文献)→周日发合作者review。")
    else:
        print("→ 补齐实验+定署名,再冲刺论文完整草稿。")

if __name__ == "__main__":
    track()
```

```
   □ 待办: 所有实验做扎实+归档

署名: 2/2 = 100%

组会汇报累计: 2 次 (M17硬交付要≥2次) ✅

==========================================
实验80% / 论文章节29% / 署名已定
→ W67冲刺完整草稿(剩余章节:Intro/RW/Conclusion/Abstract/参考文献)→周日发合作者review。
```
> 解读：把冲刺进度填进去，脚本算出实验/论文章节/署名的进度和组会汇报累计。示例里实验接近完成（差做扎实归档）、署名已定、组会汇报累计 2 次（M17 硬交付达成），论文章节才 29%（W66 写了 Method/Experiment，W67 要冲刺剩余章节）。它把"离完整草稿还差多少"一目了然——W67 集中冲刺剩余章节、周日发合作者 review。
